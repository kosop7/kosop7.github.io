    <div>⭐ SCORE: <span id="score">0</span></div>

  <canvas id="game" width="960" height="600" tabindex="0"></canvas>

<script>
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");
canvas.focus();

/* ================= INPUT ================= */
const key={l:0,r:0,u:0,d:0};
addEventListener("keydown",e=>{
  if(e.key==="ArrowLeft"||e.key==="a") key.l=1;
  if(e.key==="ArrowRight"||e.key==="d") key.r=1;
  if(e.key==="ArrowUp"||e.key==="w") key.u=1;
  if(e.key==="ArrowDown"||e.key==="s") key.d=1;
});
addEventListener("keyup",e=>{
  if(e.key==="ArrowLeft"||e.key==="a") key.l=0;
  if(e.key==="ArrowRight"||e.key==="d") key.r=0;
  if(e.key==="ArrowUp"||e.key==="w") key.u=0;
  if(e.key==="ArrowDown"||e.key==="s") key.d=0;
});

/* ================= PLAYER (BIRD) ================= */
const player={
  x:480,y:300,
  vx:0,vy:0,
  speed:3,
  hp:100
};

/* ================= ENEMIES ================= */
const enemies=[];
let spawnTimer=0;

/* ================= HUD ================= */
const hpEl=document.getElementById("hp");
const scoreEl=document.getElementById("score");
let score=0;

/* ================= UTILS ================= */
function clamp(v,a,b){ return Math.max(a,Math.min(b,v)); }

/* ================= GAME LOOP ================= */
function update(){
  /* player movement */
  player.vx=(key.r-key.l)*player.speed;
  player.vy=(key.d-key.u)*player.speed;
  player.x+=player.vx;
  player.y+=player.vy;

  player.x=clamp(player.x,20,940);
  player.y=clamp(player.y,20,580);

  /* enemy spawn */
  spawnTimer++;
  if(spawnTimer>80){
    spawnTimer=0;
    enemies.push({
      x:Math.random()*960,
      y:-20,
      r:10,
      speed:1.5+Math.random()*1.5
    });
  }

  /* enemy move + collision */
  enemies.forEach(e=>{
    const dx=player.x-e.x;
    const dy=player.y-e.y;
    const d=Math.hypot(dx,dy);
    e.x+=dx/d*e.speed;
    e.y+=dy/d*e.speed;

    if(d<18){
      player.hp-=0.4;
    }
  });

  /* cleanup */
  for(let i=enemies.length-1;i>=0;i--){
    if(enemies[i].x<-50||enemies[i].x>1010||enemies[i].y>650){
      enemies.splice(i,1);
      score++;
    }
  }

  hpEl.textContent=Math.max(0,player.hp|0);
  scoreEl.textContent=score;
}

/* ================= DRAW ================= */
function draw(){
  ctx.clearRect(0,0,960,600);

  /* grid */
  ctx.strokeStyle="rgba(0,0,0,0.04)";
  for(let x=0;x<960;x+=48){
    ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,600); ctx.stroke();
  }
  for(let y=0;y<600;y+=48){
    ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(960,y); ctx.stroke();
  }

  /* bird (player) */
  ctx.fillStyle="#00c896";
  ctx.beginPath();
  ctx.ellipse(player.x,player.y,12,8,0,0,Math.PI*2);
  ctx.fill();

  /* wing */
  ctx.strokeStyle="#00a77a";
  ctx.beginPath();
  ctx.moveTo(player.x-6,player.y);
  ctx.lineTo(player.x-18,player.y-6);
  ctx.stroke();

  /* beak */
  ctx.fillStyle="#ffb703";
  ctx.beginPath();
  ctx.moveTo(player.x+12,player.y);
  ctx.lineTo(player.x+20,player.y-4);
  ctx.lineTo(player.x+20,player.y+4);
  ctx.fill();

  /* enemies */
  ctx.fillStyle="#ff4d4d";
  enemies.forEach(e=>{
    ctx.beginPath();
    ctx.arc(e.x,e.y,e.r,0,Math.PI*2);
    ctx.fill();
  });
}

/* ================= LOOP ================= */
function loop(){
  if(player.hp>0){
    update();
    draw();
    requestAnimationFrame(loop);
  }else{
    ctx.fillStyle="rgba(255,255,255,.9)";
    ctx.fillRect(0,0,960,600);
    ctx.fillStyle="#111";
    ctx.font="28px sans-serif";
    ctx.fillText("GAME OVER",400,300);
  }
}
loop();
</script>
