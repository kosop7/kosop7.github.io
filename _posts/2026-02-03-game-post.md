<div style="display:flex; justify-content:center; margin:40px 0;">
  <canvas id="game" width="800" height="400"
    style="background:#111; border:2px solid #333;"></canvas>
</div>

<div style="text-align:center; margin-top:10px;">
  <button onclick="input.left=true">◀</button>
  <button onclick="jump()">Q</button>
  <button onclick="attack('punch')">W</button>
  <button onclick="attack('kick')">E</button>
  <button onclick="guard()">R</button>
  <button onclick="input.right=true">▶</button>
</div>

<script>
/* ===== BASIC ===== */
const c = document.getElementById("game");
const ctx = c.getContext("2d");
const GROUND = 320;

/* ===== SOUND ===== */
const sndHit = new Audio("https://assets.mixkit.co/active_storage/sfx/2165/2165-preview.mp3");
const sndKick = new Audio("https://assets.mixkit.co/active_storage/sfx/2174/2174-preview.mp3");

/* ===== INPUT ===== */
const input = {};
document.addEventListener("keydown",e=>input[e.key]=true);
document.addEventListener("keyup",e=>input[e.key]=false);

/* ===== PLAYER ===== */
const p1 = { x:200, y:GROUND, vy:0, hp:100, state:"idle", dir:1 };
const p2 = { x:600, y:GROUND, vy:0, hp:100, state:"idle", dir:-1 };

/* ===== ACTION ===== */
function jump(){
  if(p1.y===GROUND){ p1.vy=-12; }
}

function guard(){
  p1.state="guard";
  setTimeout(()=>p1.state="idle",300);
}

function attack(type){
  if(p1.state!=="idle") return;
  p1.state=type;
  (type==="punch"?sndHit:sndKick).play();

  if(Math.abs(p1.x-p2.x)<60 && p2.state!=="guard"){
    p2.hp -= type==="punch"?10:15;
    p2.state="hit";
  }

  setTimeout(()=>p1.state="idle",300);
}

/* ===== AI ===== */
function enemyAI(){
  if(p2.state!=="idle") return;
  if(p2.x>p1.x+50) p2.x-=2;
  if(p2.x<p1.x-50) p2.x+=2;

  if(Math.random()<0.02){
    p2.state="punch";
    sndHit.play();
    if(Math.abs(p2.x-p1.x)<60 && p1.state!=="guard"){
      p1.hp-=10;
    }
    setTimeout(()=>p2.state="idle",300);
  }
}

/* ===== PHYSICS ===== */
function physics(p){
  p.vy+=0.6;
  p.y+=p.vy;
  if(p.y>GROUND){p.y=GROUND;p.vy=0;}
}

/* ===== DRAW (사람 형태 유지) ===== */
function drawFighter(p,color){
  ctx.strokeStyle=color;
  ctx.lineWidth=4;

  // 머리
  ctx.beginPath();
  ctx.arc(p.x,p.y-80,10,0,Math.PI*2);
  ctx.stroke();

  // 몸통
  ctx.beginPath();
  ctx.moveTo(p.x,p.y-70);
  ctx.lineTo(p.x,p.y-35);
  ctx.stroke();

  // 팔
  ctx.beginPath();
  ctx.moveTo(p.x,p.y-55);
  ctx.lineTo(p.x+25*p.dir,p.y-45);
  ctx.stroke();

  // 다리
  ctx.beginPath();
  ctx.moveTo(p.x,p.y-35);
  ctx.lineTo(p.x+15,p.y);
  ctx.stroke();

  ctx.beginPath();
  ctx.moveTo(p.x,p.y-35);
  ctx.lineTo(p.x-15,p.y);
  ctx.stroke();
}

/* ===== LOOP ===== */
function update(){
  if(input["ArrowLeft"]) p1.x-=4;
  if(input["ArrowRight"]) p1.x+=4;
  if(input["q"]) jump();
  if(input["w"]) attack("punch");
  if(input["e"]) attack("kick");
  if(input["r"]) guard();

  physics(p1); physics(p2);
  enemyAI();

  if(p1.hp<=0 || p2.hp<=0){
    alert(p1.hp<=0 ? "패배! 다시?" : "승리! 다시?");
    location.reload();
  }
}

function draw(){
  ctx.clearRect(0,0,800,400);
  ctx.fillStyle="#555";
  ctx.fillRect(0,GROUND,800,5);

  drawFighter(p1,"#4af");
  drawFighter(p2,"#f44");

  ctx.fillStyle="#0f0";
  ctx.fillRect(50,20,p1.hp*2,10);
  ctx.fillRect(550,20,p2.hp*2,10);
}

function loop(){
  update(); draw();
  requestAnimationFrame(loop);
}
loop();
</script>
