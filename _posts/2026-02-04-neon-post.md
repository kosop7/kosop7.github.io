<script>
/* ========= canvas & focus ========= */
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");
canvas.focus();

function lockFocus(){ canvas.focus(); }
document.addEventListener("click", lockFocus);

/* ========= input ========= */
const key = {l:0,r:0,u:0,d:0,atk:0};

window.addEventListener("keydown", e=>{
  if(["ArrowUp","ArrowDown","ArrowLeft","ArrowRight"," "].includes(e.key)){
    e.preventDefault();
  }
  if(e.key==="ArrowLeft"||e.key==="a") key.l=1;
  if(e.key==="ArrowRight"||e.key==="d") key.r=1;
  if(e.key==="ArrowUp"||e.key==="w") key.u=1;
  if(e.key==="ArrowDown"||e.key==="s") key.d=1;
  if(e.key===" ") key.atk=1;
});

window.addEventListener("keyup", e=>{
  if(e.key==="ArrowLeft"||e.key==="a") key.l=0;
  if(e.key==="ArrowRight"||e.key==="d") key.r=0;
  if(e.key==="ArrowUp"||e.key==="w") key.u=0;
  if(e.key==="ArrowDown"||e.key==="s") key.d=0;
  if(e.key===" ") key.atk=0;
});

/* ========= game state ========= */
let px=480, py=300;
let score=0, hp=100;
let playing=false;

const hpEl=document.getElementById("hp");
const scoreEl=document.getElementById("score");

/* ========= game loop ========= */
function loop(){
  if(playing){
    if(key.l) px-=3;
    if(key.r) px+=3;
    if(key.u) py-=3;
    if(key.d) py+=3;
  }

  ctx.clearRect(0,0,960,600);

  /* subtle grid */
  ctx.strokeStyle="rgba(0,0,0,0.04)";
  for(let x=0;x<960;x+=48){
    ctx.beginPath(); ctx.moveTo(x,0); ctx.lineTo(x,600); ctx.stroke();
  }
  for(let y=0;y<600;y+=48){
    ctx.beginPath(); ctx.moveTo(0,y); ctx.lineTo(960,y); ctx.stroke();
  }

  /* player */
  ctx.fillStyle="#00c896";
  ctx.beginPath();
  ctx.arc(px,py,12,0,Math.PI*2);
  ctx.fill();

  hpEl.textContent=hp;
  scoreEl.textContent=score;

  requestAnimationFrame(loop);
}
loop();

/* ========= start ========= */
document.getElementById("start").onclick=()=>{
  playing=true;
  document.getElementById("overlay").style.display="none";
  canvas.focus();
};
</script>
