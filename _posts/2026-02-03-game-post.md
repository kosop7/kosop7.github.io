  <canvas id="game" width="820" height="420"
    style="background:#000; border:2px solid #444;"></canvas>

<script>
/* =====================
   CORE
===================== */
const c = document.getElementById("game");
const ctx = c.getContext("2d");
const GROUND = 330;

/* =====================
   SAVE
===================== */
let save = JSON.parse(localStorage.getItem("tekkenSave")) || { best:1 };
function saveGame(stage){
  if(stage>save.best){
    save.best=stage;
    localStorage.setItem("tekkenSave",JSON.stringify(save));
  }
}

/* =====================
   INPUT + COMMAND
===================== */
const input={};
let commandBuffer=[];

document.addEventListener("keydown",e=>{
  input[e.key]=true;

  if(["ArrowDown","ArrowRight"].includes(e.key)){
    commandBuffer.push({key:e.key,time:Date.now()});
    if(commandBuffer.length>5) commandBuffer.shift();
  }

  if(e.key==="w") attack("punch");
  if(e.key==="e") attack("kick");
});
document.addEventListener("keyup",e=>input[e.key]=false);

function checkCommand(){
  if(commandBuffer.length<2) return false;
  const a=commandBuffer.at(-2),b=commandBuffer.at(-1);
  return a.key==="ArrowDown" && b.key==="ArrowRight" && (b.time-a.time<300);
}

/* =====================
   FIGHTER
===================== */
function Fighter(x,color,dir,hp,speed){
  return{
    x,y:GROUND,vy:0,
    color,dir,
    hp,maxHp:hp,
    speed,
    state:"idle",
    frame:0
  };
}

/* =====================
   ENEMIES
===================== */
const enemyData=[
  { color:"#f44", hp:80, speed:2, rate:0.02, fav:"punch" },
  { color:"#fa4", hp:100, speed:2.5, rate:0.03, fav:"kick" },
  { color:"#f4f", hp:130, speed:3, rate:0.04, fav:"special" }
];

let stage=1;
let enemyIndex=0;
let bgColors=["#111","#112233","#220022","#331100"];

let p1 = Fighter(180,"#4af",1,100,4);
let p2;

/* =====================
   ACTIONS
===================== */
function attack(type){
  if(p1.state!=="idle") return;

  if(type==="punch"&&checkCommand()){
    type="special";
    commandBuffer=[];
  }

  p1.state=type;
  p1.frame=0;

  const dmg={punch:10,kick:14,special:25}[type];
  const range={punch:35,kick:50,special:70}[type];

  if(Math.abs(p1.x-p2.x)<range){
    p2.hp-=dmg;
    p2.state="hit";
  }
}

/* =====================
   AI
===================== */
function ai(){
  const d=enemyData[enemyIndex];
  if(p2.state!=="idle") return;

  if(p2.x>p1.x+40) p2.x-=p2.speed;
  if(p2.x<p1.x-40) p2.x+=p2.speed;

  if(Math.random()<d.rate){
    let atk=d.fav;
    if(atk==="special"&&Math.random()<0.5) atk="kick";
    p2.state=atk;
    if(Math.abs(p2.x-p1.x)<50){
      p1.hp-= atk==="special"?20:12;
      p1.state="hit";
    }
  }
}

/* =====================
   PHYSICS
===================== */
function physics(p){
  p.vy+=0.6;
  p.y+=p.vy;
  if(p.y>GROUND){p.y=GROUND;p.vy=0;}
}

/* =====================
   DRAW
===================== */
function drawFighter(p){
  ctx.strokeStyle=p.color;
  ctx.lineWidth=4;
  ctx.beginPath();ctx.arc(p.x,p.y-80,10,0,6.28);ctx.stroke();
  ctx.beginPath();ctx.moveTo(p.x,p.y-70);ctx.lineTo(p.x,p.y-35);ctx.stroke();
  ctx.beginPath();ctx.moveTo(p.x,p.y-60);
  ctx.lineTo(p.x+25*p.dir,p.y-50);ctx.stroke();
  ctx.beginPath();ctx.moveTo(p.x,p.y-35);
  ctx.lineTo(p.x+20*p.dir,p.y);ctx.stroke();
}

function hpBar(p,x){
  ctx.fillStyle="#333";ctx.fillRect(x,20,200,10);
  ctx.fillStyle="#0f0";
  ctx.fillRect(x,20,(p.hp/p.maxHp)*200,10);
}

/* =====================
   ROUND
===================== */
function loadEnemy(){
  const d=enemyData[enemyIndex];
  p2=Fighter(620,d.color,-1,d.hp+stage*10,d.speed+stage*0.3);
}

loadEnemy();

/* =====================
   LOOP
===================== */
function update(){
  if(input["ArrowLeft"]) p1.x-=p1.speed;
  if(input["ArrowRight"]) p1.x+=p1.speed;

  physics(p1); physics(p2); ai();

  [p1,p2].forEach(p=>{
    p.frame++;
    if(p.frame>18){p.state="idle";p.frame=0;}
  });

  if(p2.hp<=0){
    enemyIndex++;
    if(enemyIndex>=enemyData.length){
      stage++;
      saveGame(stage);
      enemyIndex=0;
    }
    loadEnemy();
  }

  if(p1.hp<=0){
    stage=1; enemyIndex=0;
    p1.hp=p1.maxHp;
    loadEnemy();
  }
}

function draw(){
  ctx.fillStyle=bgColors[(stage-1)%bgColors.length];
  ctx.fillRect(0,0,820,420);
  ctx.fillStyle="#555";ctx.fillRect(0,GROUND,820,5);

  drawFighter(p1); drawFighter(p2);
  hpBar(p1,60); hpBar(p2,560);

  ctx.fillStyle="#fff";
  ctx.fillText("STAGE "+stage+"  BEST "+save.best,360,15);
}

function loop(){update();draw();requestAnimationFrame(loop);}
loop();
</script>
