<div style="text-align:center; margin-bottom:40px;">
  <button onmousedown="keys.l=true" onmouseup="keys.l=false">◀</button>
  <button onmousedown="keys.r=true" onmouseup="keys.r=false">▶</button>
  <button onclick="jump()">JUMP</button>
  <button onclick="attack('punch')">PUNCH</button>
  <button onclick="attack('kick')">KICK</button>
  <button onclick="guard()">GUARD</button>
</div>

<script>
/* =========================
   CORE SETUP
========================= */
const c = document.getElementById("tekken");
const ctx = c.getContext("2d");
const GROUND = 330;

const keys = { l:false, r:false };
document.addEventListener("keydown", e=>{
  if(e.key==="a") keys.l=true;
  if(e.key==="d") keys.r=true;
});
document.addEventListener("keyup", e=>{
  if(e.key==="a") keys.l=false;
  if(e.key==="d") keys.r=false;
});

/* =========================
   AUDIO (NO FILES)
========================= */
const audio = new (window.AudioContext||window.webkitAudioContext)();
function hitSound(freq=180){
  const o=audio.createOscillator();
  o.frequency.value=freq;
  o.connect(audio.destination);
  o.start();
  o.stop(audio.currentTime+0.04);
}

/* =========================
   FIGHTER OBJECT
========================= */
function Fighter(x,color,dir){
  return {
    x, y:GROUND, vy:0,
    hp:100,
    dir, color,
    state:"idle", // idle, punch, kick, guard, hit
    frame:0,
    combo:0
  };
}

const p1 = Fighter(180,"#4af", 1);
const p2 = Fighter(620,"#f44",-1);

/* =========================
   CONTROLS
========================= */
function jump(){
  if(p1.y===GROUND) p1.vy=-13;
}

function guard(){
  p1.state="guard";
  p1.frame=0;
}

function attack(type){
  if(p1.state!=="idle") return;
  p1.state=type;
  p1.frame=0;
  p1.combo=(p1.combo+1)%3;

  const range = type==="kick"?50:35;
  if(Math.abs(p1.x-p2.x)<range){
    if(p2.state!=="guard"){
      p2.hp -= type==="kick"?14:10;
      p2.state="hit";
      hitSound(type==="kick"?120:200);
    }
  }
}

/* =========================
   AI (AGGRESSIVE)
========================= */
function ai(){
  if(p2.state!=="idle") return;

  if(p2.x>p1.x+45) p2.x-=2;
  if(p2.x<p1.x-45) p2.x+=2;

  if(Math.random()<0.02){
    p2.state=Math.random()<0.5?"punch":"kick";
    if(Math.abs(p2.x-p1.x)<40){
      if(p1.state!=="guard"){
        p1.hp-=12;
        p1.state="hit";
        hitSound();
      }
    }
  }
}

/* =========================
   PHYSICS
========================= */
function physics(p){
  p.vy+=0.6;
  p.y+=p.vy;
  if(p.y>GROUND){ p.y=GROUND; p.vy=0; }
}

/* =========================
   DRAW HUMAN FIGHTER
========================= */
function drawFighter(p){
  const cx=p.x, cy=p.y;
  ctx.strokeStyle=p.color;
  ctx.lineWidth=4;

  // head
  ctx.beginPath();
  ctx.arc(cx, cy-80, 10, 0, Math.PI*2);
  ctx.stroke();

  // body
  ctx.beginPath();
  ctx.moveTo(cx, cy-70);
  ctx.lineTo(cx, cy-35);
  ctx.stroke();

  // arms
  ctx.beginPath();
  ctx.moveTo(cx, cy-60);
  if(p.state==="punch")
    ctx.lineTo(cx+25*p.dir, cy-55);
  else if(p.state==="guard")
    ctx.lineTo(cx+15*p.dir, cy-50);
  else
    ctx.lineTo(cx+18*p.dir, cy-50);
  ctx.stroke();

  // legs
  ctx.beginPath();
  ctx.moveTo(cx, cy-35);
  if(p.state==="kick")
    ctx.lineTo(cx+25*p.dir, cy-10);
  else
    ctx.lineTo(cx+10, cy);
  ctx.stroke();

  ctx.beginPath();
  ctx.moveTo(cx, cy-35);
  ctx.lineTo(cx-10, cy);
  ctx.stroke();
}

/* =========================
   UI
========================= */
function hpBar(p,x){
  ctx.fillStyle="#333";
  ctx.fillRect(x,20,200,10);
  ctx.fillStyle="#0f0";
  ctx.fillRect(x,20,Math.max(0,p.hp)*2,10);
}

/* =========================
   MAIN LOOP
========================= */
function update(){
  if(keys.l) p1.x-=4;
  if(keys.r) p1.x+=4;

  physics(p1);
  physics(p2);
  ai();

  [p1,p2].forEach(p=>{
    p.frame++;
    if(p.frame>18){
      if(p.state!=="idle") p.state="idle";
      p.frame=0;
    }
  });
}

function draw(){
  ctx.clearRect(0,0,820,420);

  ctx.fillStyle="#555";
  ctx.fillRect(0,GROUND,820,5);

  drawFighter(p1);
  drawFighter(p2);

  hpBar(p1,60);
  hpBar(p2,560);

  if(p1.hp<=0||p2.hp<=0){
    ctx.fillStyle="#fff";
    ctx.font="32px sans-serif";
    ctx.fillText("K.O.",380,210);
  }
}

function loop(){
  update();
  draw();
  requestAnimationFrame(loop);
}
loop();
</script>
