  <canvas id="game" width="960" height="540" tabindex="0"
    style="background:#000; border-radius:14px;"></canvas>
</div>

<script>
/* =====================
   AT THE EDGE OF EVENING
===================== */

const c = document.getElementById('game');
const ctx = c.getContext('2d');
c.focus();

/* ====== SAVE ====== */
const saveKey = 'seriesSave_v2';
const save = JSON.parse(localStorage.getItem(saveKey)) || {
  episode: 1,
  talked: {},
  visited: { village: true }
};
function saveGame(){ localStorage.setItem(saveKey, JSON.stringify(save)); }

/* ====== STATE ====== */
let state = 'title';
let fade = 0;
let fadeDir = -1;

/* ====== WORLD ====== */
const world = { w: 2600, h: 1600 };
let region = 'village';
const regions = {
  village: { name: 'THE VILLAGE AT DUSK', ground:'#2b2926', sky1:'#6b88a6', sky2:'#14141a' },
  gate:    { name: 'THE ROAD BEYOND',      ground:'#161616', sky1:'#30323a', sky2:'#000' }
};

/* ====== HERO ====== */
const hero = { x:1200, y:720, speed:2.2, dir:1, state:'idle' };

/* ====== CAMERA ====== */
const cam = { x:0, y:0 };

/* ====== NPC ====== */
const npcs = [
  { id:'old', x:900, y:780, region:'village', name:'노인',
    text:['해가 지면 길이 잘 안 보이지.','그래서 우린 멈추지.'] },
  { id:'child', x:1400, y:680, region:'village', name:'아이',
    text:['밖엔 어떤 곳이냐고?','나는 아직 잘 몰라.'] }
];

/* ====== INPUT ====== */
const keys = {};
let dialog = null;

function startGame(){
  if(state !== 'title') return;
  state = 'play';
}

document.addEventListener('keydown', e=>{
  keys[e.key] = true;
  startGame();
});
document.addEventListener('keyup', e=> keys[e.key]=false);
c.addEventListener('click', ()=>{
  c.focus();
  startGame();
});

/* ====== UTILS ====== */
function clamp(v,a,b){ return Math.max(a, Math.min(b,v)); }

/* ====== UPDATE ====== */
let time = 0;
function update(){
  time += 0.03;
  if(state !== 'play') return;

  let moving = false;
  if(!dialog){
    if(keys['ArrowLeft']){ hero.x-=hero.speed; hero.dir=-1; moving=true; }
    if(keys['ArrowRight']){ hero.x+=hero.speed; hero.dir=1; moving=true; }
    if(keys['ArrowUp']){ hero.y-=hero.speed; moving=true; }
    if(keys['ArrowDown']){ hero.y+=hero.speed; moving=true; }
  }
  hero.x = clamp(hero.x, 80, world.w-80);
  hero.y = clamp(hero.y, 220, world.h-80);
  hero.state = moving ? 'walk' : 'idle';

  cam.x += (hero.x - cam.x - c.width/2)*0.08;
  cam.y += (hero.y - cam.y - c.height/2)*0.08;
}

/* ====== DRAW HERO (도형 용사) ====== */
function drawHero(){
  const x = hero.x - cam.x;
  const y = hero.y - cam.y;
  const sway = Math.sin(time*0.15)*6*hero.dir;
  const breath = Math.sin(time*0.08)*1.5;

  // shadow
  ctx.fillStyle='rgba(0,0,0,0.25)';
  ctx.beginPath(); ctx.ellipse(x,y+12,18,8,0,0,6.28); ctx.fill();

  // cloak
  ctx.fillStyle='#f0c94a';
  ctx.beginPath();
  ctx.moveTo(x-18+sway*0.2,y-10+breath);
  ctx.lineTo(x+10+sway*0.6,y-10+breath);
  ctx.lineTo(x+24+sway,y+28+breath);
  ctx.lineTo(x-28+sway*0.1,y+28+breath);
  ctx.closePath(); ctx.fill();

  // body
  ctx.fillStyle='#3b3b3b';
  ctx.fillRect(x-8,y-6+breath,16,20);

  // head
  ctx.fillStyle='#efe8db';
  ctx.beginPath(); ctx.arc(x+hero.dir*2,y-22,9,0,6.28); ctx.fill();

  // eyes
  ctx.fillStyle='#222';
  ctx.beginPath(); ctx.arc(x-3,y-24,1.5,0,6.28); ctx.fill();
  ctx.beginPath(); ctx.arc(x+3,y-24,1.5,0,6.28); ctx.fill();
}

/* ====== DRAW ====== */
function draw(){
  // sky
  const g = ctx.createLinearGradient(0,0,0,c.height);
  g.addColorStop(0, regions[region].sky1);
  g.addColorStop(1, regions[region].sky2);
  ctx.fillStyle=g;
  ctx.fillRect(0,0,c.width,c.height);

  // ground
  ctx.fillStyle=regions[region].ground;
  ctx.fillRect(-cam.x,c.height-160-cam.y,world.w,200);

  // hero
  if(state!=='title') drawHero();

  // title overlay
  if(state==='title'){
    ctx.fillStyle='rgba(0,0,0,0.6)';
    ctx.fillRect(0,0,c.width,c.height);
    ctx.fillStyle='#fff';
    ctx.font='30px serif';
    ctx.fillText('AT THE EDGE OF EVENING', c.width/2-200, c.height/2-10);
    ctx.font='14px serif';
    ctx.fillText('Click or Press Any Key', c.width/2-90, c.height/2+20);
  }
}

/* ====== LOOP ====== */
function loop(){
  update();
  draw();
  requestAnimationFrame(loop);
}
loop();
</script>
