<script>
/* =====================
   AT THE EDGE OF EVENING — refined hero (shape-based)
   Single-file demo — paste into GitHub Pages or Jekyll post
   Controls: Arrow keys / Space (interact) / 1/2 (dialog choices)
===================== */

const c = document.getElementById('game');
const ctx = c.getContext('2d');

/* ====== SAVE ====== */
const saveKey = 'seriesSave_v2';
const save = JSON.parse(localStorage.getItem(saveKey)) || {
  episode: 1,
  talked: {},
  visited: { village: true }
};
function saveGame(){ localStorage.setItem(saveKey, JSON.stringify(save)); }

/* ====== STATE ====== */
let state = 'title'; // title | play | transition | end
let fade = 1;        // for transitions (0..1)
let fadeDir = -1;

/* ====== WORLD & REGION ====== */
const world = { w: 2600, h: 1600 };
let region = 'village';
const regions = {
  village: { name: 'THE VILLAGE AT DUSK', ground:'#2b2926', sky1:'#6b88a6', sky2:'#14141a' },
  gate:    { name: 'THE ROAD BEYOND',      ground:'#161616', sky1:'#30323a', sky2:'#000' }
};

/* ====== PLAYER (hero built from shapes) ====== */
const hero = {
  x: 1200,
  y: 720,
  speed: 2.2,
  vx: 0, vy: 0,
  dir: 1, // facing (1 right, -1 left)
  state: 'idle', // idle / walk / talk
  hp: 100
};

/* ====== CAMERA (smooth) ====== */
const cam = { x: 0, y: 0 };

/* ====== NPCs ====== */
const npcs = [
  { id:'old', x:900, y:780, region:'village',
    name:'노인', text:['해가 지면 길이 잘 안 보이지.','그래서 우린 멈추지.'] },
  { id:'child', x:1400, y:680, region:'village',
    name:'아이', text:['밖엔 어떤 곳이냐고?','나는 아직 잘 몰라.'] }
];

/* ====== INPUT ====== */
const keys = {};
let startedAudio = false;
document.addEventListener('keydown', e=>{
  keys[e.key] = true;
  if(state === 'title') {
    state = 'play';
    fadeDir = -1;
    if(!startedAudio) initAudio();
  }
});
document.addEventListener('keyup', e=>{ keys[e.key] = false; });

/* ====== TIME & WEATHER ====== */
let time = 0;
let weather = Math.random() < 0.35 ? 'rain' : Math.random() < 0.5 ? 'fog' : 'clear';

/* ====== AUDIO (WebAudio; starts on first user gesture) ====== */
let audioCtx = null, ambientGain = null;
function initAudio(){
  try {
    audioCtx = new (window.AudioContext || window.webkitAudioContext)();
    // ambient drone
    const osc = audioCtx.createOscillator();
    const gain = audioCtx.createGain();
    osc.type = 'sine';
    osc.frequency.value = 60;
    gain.gain.value = 0.02;
    osc.connect(gain);
    gain.connect(audioCtx.destination);
    osc.start();
    ambientGain = gain;
    startedAudio = true;
  } catch(e){
    console.warn('Audio init failed', e);
  }
}
function playHit(freq = 300, dur = 0.06){
  if(!audioCtx) return;
  const o = audioCtx.createOscillator();
  const g = audioCtx.createGain();
  o.type = 'square';
  o.frequency.value = freq;
  g.gain.value = 0.08;
  o.connect(g); g.connect(audioCtx.destination);
  o.start();
  g.gain.exponentialRampToValueAtTime(0.0001, audioCtx.currentTime + dur);
  o.stop(audioCtx.currentTime + dur);
}

/* ====== UTILS ====== */
function clamp(v,a,b){ return Math.max(a, Math.min(b, v)); }

/* ====== CAMERA TARGET CALC ====== */
function updateCamera(){
  const tx = clamp(hero.x - c.width/2, 0, world.w - c.width);
  const ty = clamp(hero.y - c.height/2, 0, world.h - c.height);
  cam.x += (tx - cam.x) * 0.08;
  cam.y += (ty - cam.y) * 0.08;
}

/* ====== HERO PHYSICS & INPUT ====== */
function updateHero(){
  if(state !== 'play') return;

  let moving = false;
  if(!dialog){ // only move when not in dialog
    if(keys['ArrowLeft'] || keys['a']) { hero.x -= hero.speed; hero.dir = -1; moving = true; }
    if(keys['ArrowRight'] || keys['d']) { hero.x += hero.speed; hero.dir = 1; moving = true; }
    if(keys['ArrowUp'] || keys['w']) { hero.y -= hero.speed; moving = true; }
    if(keys['ArrowDown'] || keys['s']) { hero.y += hero.speed; moving = true; }
    // clamp to world
    hero.x = clamp(hero.x, 80, world.w-80);
    hero.y = clamp(hero.y, 220, world.h-80);
  }

  hero.state = moving ? 'walk' : 'idle';
}

/* ====== DIALOG ====== */
let dialog = null;
let dialogCooldown = 0;

/* ====== TRIGGER DIALOG ====== */
function tryInteract(){
  if(state !== 'play' || dialog || dialogCooldown > 0) return;
  for(const n of npcs){
    if(n.region !== region) continue;
    const d = Math.hypot(hero.x - n.x, hero.y - n.y);
    if(d < 36){
      dialog = { npc: n, line: 0 };
      save.talked[n.id] = true;
      saveGame();
      playHit(420, 0.08);
      dialogCooldown = 12;
      break;
    }
  }
}

/* space interaction */
document.addEventListener('keydown', e=>{
  if(e.key === ' ') { tryInteract(); e.preventDefault(); }
  if(dialog && (e.key === '1' || e.key === '2')){
    // choice controls: 1 advance, 2 dismiss
    if(e.key === '1'){ dialog.line++; if(dialog.line >= dialog.npc.text.length) dialog = null; }
    else { dialog = null; }
  }
});

/* ====== REGION TRANSITION (gate) ====== */
function checkRegionTransition(){
  if(region === 'village' && hero.x > 2400){
    state = 'transition';
    fade = 0;
    fadeDir = 1;
    // mark next region locked/visited
  }
  if(state === 'transition'){
    fade += fadeDir * 0.02;
    if(fade >= 1){
      region = 'gate';
      save.visited.gate = true;
      saveGame();
      // reset hero position a bit
      hero.x = 200;
      fadeDir = -1;
    }
    if(fade <= 0){ state = 'play'; fade = 0; }
  }
}

/* ====== RENDER HELPERS ====== */
function drawSky(){
  const r = regions[region];
  const t = clamp((Math.sin(time*0.02)+1)/2, 0, 1) * 0.9;
  const g = ctx.createLinearGradient(0,0,c.width, c.height);
  g.addColorStop(0, r.sky1);
  g.addColorStop(1, r.sky2);
  ctx.fillStyle = g;
  ctx.fillRect(0,0,c.width, c.height);
}

function drawParallaxBackground(){
  // simple layered hills for depth
  const baseX = -cam.x * 0.2;
  for(let i=0;i<3;i++){
    const hue = i===0 ? '#334' : (i===1? '#253' : '#1b2');
    ctx.fillStyle = hue;
    const offsetY = 200 + i*40;
    ctx.beginPath();
    ctx.moveTo(-100 + baseX * (0.6+i*0.2), c.height - offsetY);
    for(let x = -100; x < c.width + 200; x += 80){
      const y = c.height - offsetY - Math.sin((x + baseX) * 0.004 + i) * (20 + i*8);
      ctx.quadraticCurveTo(x + 40, y - 20, x + 80, c.height - offsetY);
    }
    ctx.lineTo(c.width + 200, c.height);
    ctx.lineTo(-100, c.height);
    ctx.fill();
  }
}

function drawGround(){
  const r = regions[region];
  ctx.fillStyle = r.ground;
  ctx.fillRect(-cam.x, c.height - 160 - cam.y, world.w, 200);
  // subtle tile pattern
  ctx.fillStyle = 'rgba(255,255,255,0.02)';
  for(let x = Math.floor(( -cam.x )/20)*20; x < -cam.x + c.width; x += 40){
    for(let y = c.height - 160 - cam.y; y < c.height - cam.y; y += 40){
      ctx.fillRect(x + 4, y + 4, 18, 18);
    }
  }
}

/* ====== WEATHER PARTICLES (rain/fog) ====== */
const rainDrops = [];
for(let i=0;i<200;i++){
  rainDrops.push({
    x: Math.random()*world.w,
    y: Math.random()*world.h,
    len: 6 + Math.random()*10,
    speed: 2 + Math.random()*2
  });
}
function updateWeather(){
  if(weather === 'rain'){
    for(const r of rainDrops){
      r.y += r.speed + 2;
      r.x += -0.5; // wind
      if(r.y > world.h) { r.y = -20; r.x = Math.random()*world.w; }
    }
  }
}

/* ====== DRAW HERO (shape-based) ====== */
function drawHero(h){
  const sx = Math.round(h.x - cam.x);
  const sy = Math.round(h.y - cam.y);

  // breathing / bob
  const breath = Math.sin(time*0.08) * 1.6;

  // cloak sway
  const sway = Math.sin(time*0.15) * 6 * (h.dir);

  // shadow
  ctx.fillStyle = 'rgba(0,0,0,0.25)';
  ctx.beginPath();
  ctx.ellipse(sx, sy + 12, 18, 8, 0, 0, Math.PI*2);
  ctx.fill();

  // cloak (trapezoid) - yellow, slightly asymmetric
  ctx.fillStyle = '#f0c94a';
  ctx.beginPath();
  ctx.moveTo(sx - 18 + sway*0.2, sy - 10 + breath);
  ctx.lineTo(sx + 10 + sway*0.6, sy - 10 + breath);
  ctx.lineTo(sx + 24 + sway, sy + 28 + breath);
  ctx.lineTo(sx - 28 + sway*0.1, sy + 28 + breath);
  ctx.closePath();
  ctx.fill();

  // body rectangle under cloak
  ctx.fillStyle = '#3b3b3b';
  ctx.fillRect(sx - 8, sy - 6 + breath, 16, 20);

  // head (circle) - slightly forward lean
  ctx.fillStyle = '#efe8db';
  ctx.beginPath();
  ctx.arc(sx + (h.dir * 2), sy - 22 + breath*0.3, 9, 0, Math.PI*2);
  ctx.fill();
  ctx.strokeStyle = '#6b5f4a';
  ctx.lineWidth = 1.2;
  ctx.stroke();

  // eyes (two small circles) - blinking
  const blink = Math.max(0, Math.sin(time*0.12) > 0.9 ? Math.cos(time*0.12)*0.3 : 1);
  ctx.fillStyle = '#2b2b2b';
  ctx.beginPath();
  ctx.ellipse(sx - 3, sy - 24 + breath*0.2, 1.6, 1.6*blink, 0, 0, Math.PI*2);
  ctx.fill();
  ctx.beginPath();
  ctx.ellipse(sx + 3, sy - 24 + breath*0.2, 1.6, 1.6*blink, 0, 0, Math.PI*2);
  ctx.fill();

  // arms (simple line sticks) with slight rotation when walking
  ctx.strokeStyle = '#3b3b3b';
  ctx.lineWidth = 3;
  ctx.beginPath();
  const armSwing = Math.sin(time*0.2) * 10;
  ctx.moveTo(sx - 6, sy - 6 + breath);
  ctx.lineTo(sx - 16 + (h.dir*-1)*2, sy + 2 + breath + (h.state === 'walk' ? armSwing*0.2 : 0));
  ctx.stroke();

  ctx.beginPath();
  ctx.moveTo(sx + 6, sy - 6 + breath);
  ctx.lineTo(sx + 16 + (h.dir)*2, sy + 2 + breath - (h.state === 'walk' ? armSwing*0.2 : 0));
  ctx.stroke();

  // legs (short, chunky)
  ctx.lineWidth = 3.5;
  ctx.beginPath();
  ctx.moveTo(sx - 6, sy + 14 + breath);
  ctx.lineTo(sx - 6 + (h.state === 'walk' ? Math.sin(time*0.2)*6 : 0), sy + 28 + breath);
  ctx.stroke();

  ctx.beginPath();
  ctx.moveTo(sx + 6, sy + 14 + breath);
  ctx.lineTo(sx + 6 - (h.state === 'walk' ? Math.sin(time*0.2)*6 : 0), sy + 28 + breath);
  ctx.stroke();

  // small detail: hood highlight
  ctx.strokeStyle = 'rgba(255,255,255,0.08)';
  ctx.lineWidth = 1;
  ctx.beginPath();
  ctx.moveTo(sx - 10, sy - 14);
  ctx.quadraticCurveTo(sx, sy - 30, sx + 10, sy - 14);
  ctx.stroke();
}

/* ====== DRAW NPCs (small shape + subtle bob) ====== */
function drawNPC(n){
  const bx = Math.round(n.x - cam.x);
  const by = Math.round(n.y - cam.y);
  const bob = Math.sin(time*0.12 + n.x) * 1.6;
  ctx.fillStyle = '#e6e6e6';
  ctx.beginPath();
  ctx.arc(bx, by + bob, 6, 0, Math.PI*2);
  ctx.fill();
  // tiny tag
  ctx.fillStyle = 'rgba(255,255,255,0.7)';
  ctx.font = '11px serif';
  ctx.fillText(n.name, bx - 10, by - 10 + bob);
}

/* ====== DRAW FRAME ====== */
function drawFrame(){
  // sky
  drawSky();
  // parallax
  drawParallaxBackground();
  // ground layer
  drawGround();

  // weather overlay (rain lines in world coordinates)
  if(weather === 'rain'){
    ctx.strokeStyle = 'rgba(180,200,255,0.18)';
    ctx.lineWidth = 1;
    // draw a subset within camera to reduce cost
    for(let i=0;i<120;i++){
      const rx = (Math.floor((cam.x) / 20) * 20) + (i*23) % (c.width) + (Math.sin(i+time*0.1)*6);
      const ry = (i*7 + (time*1.2)) % c.height;
      ctx.beginPath();
      ctx.moveTo(rx, ry);
      ctx.lineTo(rx + 4, ry + 12);
      ctx.stroke();
    }
  }
  if(weather === 'fog'){
    ctx.fillStyle = 'rgba(220,220,220,0.06)';
    ctx.fillRect(0,0,c.width,c.height);
  }

  // draw NPCs for region
  for(const n of npcs){
    if(n.region !== region) continue;
    drawNPC(n);
  }

  // draw hero
  drawHero(hero);

  // UI: region and time
  ctx.fillStyle = 'rgba(255,255,255,0.85)';
  ctx.font = '13px serif';
  ctx.fillText(regions[region].name, 20, 28);
  ctx.fillText('time: ' + Math.floor(time), 20, 46);

  // dialog box
  if(dialog){
    ctx.fillStyle = 'rgba(0,0,0,0.78)';
    ctx.fillRect(80, c.height - 140, 800, 110);
    ctx.fillStyle = '#fff';
    ctx.font = '15px serif';
    ctx.fillText(dialog.npc.name + ': ' + dialog.npc.text[dialog.line], 110, c.height - 98);
    ctx.fillStyle = 'rgba(255,255,255,0.6)';
    ctx.font = '12px serif';
    ctx.fillText('1: 계속   2: 닫기', 110, c.height - 60);
  }

  // fade overlay for transitions/end
  if(fade > 0){
    ctx.fillStyle = 'rgba(0,0,0,' + fade + ')';
    ctx.fillRect(0,0,c.width,c.height);
    if(fade > 0.92){
      ctx.fillStyle = '#fff';
      ctx.font = '18px serif';
      ctx.fillText(regions[region].name, c.width/2 - 90, c.height/2 - 10);
    }
  }

  // ending message
  if(state === 'end'){
    ctx.fillStyle = 'rgba(0,0,0,0.6)';
    ctx.fillRect(0,0,c.width,c.height);
    ctx.fillStyle = '#fff';
    ctx.font = '20px serif';
    ctx.fillText('You stayed long enough.', c.width/2 - 120, c.height/2 - 10);
    ctx.font = '14px serif';
    ctx.fillText('Episode 2 is waiting beyond the road.', c.width/2 - 136, c.height/2 + 20);
  }
}

/* ====== GAME LOOP ====== */
function update(){
  time += 0.03;
  if(state === 'title') return;

  // dialog cooldown
  if(dialogCooldown > 0) dialogCooldown--;

  // hero movement and interactions
  updateHero();

  // region transition check
  checkRegionTransition();

  // camera
  updateCamera();

  // weather update
  updateWeather();

  // simple end condition: if visited gate and talked to both NPCs -> end after some time
  if(save.visited.gate && Object.keys(save.talked).length >= 2 && time > 500){
    state = 'end';
  }
}

function loop(){
  update();
  drawFrame();
  requestAnimationFrame(loop);
}

/* ====== INIT TITLE + Interaction to start ====== */
function drawInitialTitle(){
  ctx.fillStyle = '#000';
  ctx.fillRect(0,0,c.width,c.height);
  ctx.fillStyle = '#efecec';
  ctx.font = '30px serif';
  ctx.fillText('AT THE EDGE OF EVENING', c.width/2 - 200, c.height/2 - 10);
  ctx.font = '14px serif';
  ctx.fillText('Press any key to begin', c.width/2 - 80, c.height/2 + 20);
}

/* ====== START ====== */
drawInitialTitle();
loop();

/* ====== small helper to trigger transition manually if needed ====== */
function triggerTransitionToGate(){
  state = 'transition';
  fade = 0;
  fadeDir = 1;
}

/* ====== expose for debug in console (optional) ====== */
window.__EDGE = {
  save, saveGame, triggerTransitionToGate
};

</script>
