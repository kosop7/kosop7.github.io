          ontouchend="input.left=false"
          onmousedown="input.left=true"
          onmouseup="input.left=false">◀</button>

  <button ontouchstart="input.right=true;event.preventDefault()"
          ontouchend="input.right=false"
          onmousedown="input.right=true"
          onmouseup="input.right=false">▶</button>

  <button ontouchstart="jump();event.preventDefault()" onclick="jump()">Q JUMP</button>
  <button ontouchstart="attack('punch');event.preventDefault()" onclick="attack('punch')">W PUNCH</button>
  <button ontouchstart="attack('kick');event.preventDefault()" onclick="attack('kick')">E KICK</button>
  <button ontouchstart="guard();event.preventDefault()" onclick="guard()">R GUARD</button>

<script>
/* =====================
   GLOBAL SAFETY
===================== */
document.addEventListener("touchmove", e => {
  e.preventDefault();
}, { passive:false });

/* =====================
   CORE
===================== */
const canvas = document.getElementById("tekken");
const ctx = canvas.getContext("2d");
const GROUND = 330;

/* =====================
   INPUT
===================== */
const input = { left:false, right:false };

document.addEventListener("keydown", e=>{
  if(e.key==="ArrowLeft"||e.key==="a") input.left=true;
  if(e.key==="ArrowRight"||e.key==="d") input.right=true;
  if(e.key==="q") jump();
  if(e.key==="w") attack("punch");
  if(e.key==="e") attack("kick");
  if(e.key==="r") guard();
  if(e.key==="Enter" && (gameState==="win"||gameState==="lose")) resetGame();
});

document.addEventListener("keyup", e=>{
  if(e.key==="ArrowLeft"||e.key==="a") input.left=false;
  if(e.key==="ArrowRight"||e.key==="d") input.right=false;
});

/* =====================
   AUDIO (SAFE)
===================== */
let audio;
document.addEventListener("pointerdown", ()=>{
  if(!audio){
    audio = new (window.AudioContext||window.webkitAudioContext)();
  }
},{once:true});

function hitSound(freq=180){
  if(!audio) return;
  const o = audio.createOscillator();
  o.frequency.value = freq;
  o.connect(audio.destination);
  o.start();
  o.stop(audio.currentTime + 0.05);
}

/* =====================
   FIGHTER
===================== */
function Fighter(x,color,dir,hp=100){
  return {
    x, y:GROUND, vy:0,
    hp, maxHp:hp,
    dir, color,
    state:"idle", frame:0
  };
}

/* =====================
   ENEMIES
===================== */
const enemies = [
  { color:"#f44", hp:80 },
  { color:"#fa4", hp:100 },
  { color:"#f4f", hp:120 }
];
let enemyIndex = 0;

/* =====================
   GAME STATE
===================== */
let gameState = "fight";

let p1 = Fighter(180,"#4af",1,100);
let p2 = Fighter(620,enemies[0].color,-1,enemies[0].hp);

/* =====================
   ACTIONS
===================== */
function jump(){
  if(gameState!=="fight") return;
  if(p1.y===GROUND) p1.vy=-13;
}

function guard(){
  if(gameState!=="fight") return;
  if(p1.state==="idle"){
    p1.state="guard";
    p1.frame=0;
  }
}

function attack(type){
  if(gameState!=="fight") return;
  if(p1.state!=="idle") return;

  p1.state=type;
  p1.frame=0;

  const range = type==="kick"?50:35;
  if(Math.abs(p1.x-p2.x)<range){
    if(p2.state!=="guard"){
      p2.hp -= type==="kick"?14:10;
      p2.state="hit";
      hitSound(type==="kick"?120:200);
    }
  }
}

/* =====================
   AI
===================== */
function ai(){
  if(gameState!=="fight") return;
  if(p2.state!=="idle") return;

  if(p2.x>p1.x+45) p2.x-=2;
  if(p2.x<p1.x-45) p2.x+=2;

  if(Math.random()<0.02){
    p2.state=Math.random()<0.5?"punch":"kick";
    if(Math.abs(p2.x-p1.x)<40){
      if(p1.state!=="guard"){
        p1.hp-=12;
        p1.state="hit";
        hitSound();
      }
    }
  }
}

/* =====================
   PHYSICS
===================== */
function physics(p){
  p.vy+=0.6;
  p.y+=p.vy;
  if(p.y>GROUND){ p.y=GROUND; p.vy=0; }
}

/* =====================
   DRAW
===================== */
function drawFighter(p){
  const x=p.x,y=p.y;
  ctx.strokeStyle=p.color;
  ctx.lineWidth=4;

  ctx.beginPath(); ctx.arc(x,y-80,10,0,Math.PI*2); ctx.stroke();
  ctx.beginPath(); ctx.moveTo(x,y-70); ctx.lineTo(x,y-35); ctx.stroke();

  ctx.beginPath(); ctx.moveTo(x,y-60);
  ctx.lineTo(x+(p.state==="punch"?26:18)*p.dir,y-50);
  ctx.stroke();

  ctx.beginPath(); ctx.moveTo(x,y-35);
  ctx.lineTo(x+(p.state==="kick"?26:10)*p.dir,y);
  ctx.stroke();
}

function hpBar(p,x){
  ctx.fillStyle="#333"; ctx.fillRect(x,20,200,10);
  ctx.fillStyle="#0f0";
  ctx.fillRect(x,20,(p.hp/p.maxHp)*200,10);
}

/* =====================
   ROUND
===================== */
function nextEnemy(){
  enemyIndex++;
  if(enemyIndex>=enemies.length){
    gameState="win";
  }else{
    p2 = Fighter(620,enemies[enemyIndex].color,-1,enemies[enemyIndex].hp);
  }
}

function resetGame(){
  enemyIndex=0;
  p1 = Fighter(180,"#4af",1,100);
  p2 = Fighter(620,enemies[0].color,-1,enemies[0].hp);
  gameState="fight";
}

/* =====================
   LOOP
===================== */
function update(){
  if(gameState==="fight"){
    if(input.left) p1.x-=4;
    if(input.right) p1.x+=4;

    physics(p1); physics(p2); ai();

    [p1,p2].forEach(p=>{
      p.frame++;
      if(p.frame>18){
        p.state="idle";
        p.frame=0;
      }
    });

    if(p2.hp<=0) nextEnemy();
    if(p1.hp<=0) gameState="lose";
  }
}

function draw(){
  ctx.clearRect(0,0,820,420);
  ctx.fillStyle="#555"; ctx.fillRect(0,GROUND,820,5);

  drawFighter(p1);
  drawFighter(p2);
  hpBar(p1,60);
  hpBar(p2,560);

  if(gameState==="win"||gameState==="lose"){
    ctx.fillStyle="rgba(0,0,0,0.7)";
    ctx.fillRect(0,0,820,420);
    ctx.fillStyle="#fff";
    ctx.font="28px sans-serif";
    ctx.fillText(gameState==="win"?"YOU WIN!":"YOU LOSE",320,180);
    ctx.font="16px sans-serif";
    ctx.fillText("ENTER : 재시작",330,220);
  }
}

function loop(){
  update();
  draw();
  requestAnimationFrame(loop);
}
loop();
</script>
