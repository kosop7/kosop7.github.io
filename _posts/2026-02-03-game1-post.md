canvas{
  max-width:100%;
  border-radius:14px;
  touch-action:none;
}
</style>

<div style="display:flex;justify-content:center;margin:40px 0;">
  <canvas id="game" width="960" height="540"></canvas>
</div>

<script>
/* =========================
   A SMALL WORLD – EP.1
========================= */

/* ---------- CORE ---------- */
const c = document.getElementById("game");
const ctx = c.getContext("2d");

/* ---------- MOBILE SAFETY ---------- */
document.addEventListener("touchmove",e=>e.preventDefault(),{passive:false});

/* ---------- SAVE ---------- */
const SAVE_KEY="smallworld_ep1";
const save = JSON.parse(localStorage.getItem(SAVE_KEY)) || {
  talked:{},
  flags:{},
  region:"village"
};
function saveGame(){localStorage.setItem(SAVE_KEY,JSON.stringify(save));}

/* ---------- STATE ---------- */
let state="title";
let dialog=null;
let choiceIndex=0;

/* ---------- WORLD ---------- */
const world={w:5200,h:2600};
let region=save.region;

const regions={
  village:{ sky1:"#6b88a6", sky2:"#14141a", ground:"#2b2926", name:"THE VILLAGE" },
  forest:{ sky1:"#223322", sky2:"#000", ground:"#1a241a", name:"THE FOREST" },
  gate:{ sky1:"#2a2a2a", sky2:"#000", ground:"#111", name:"THE GATE" }
};

/* ---------- HERO ---------- */
const hero={x:1200,y:1400,speed:2.4,dir:1};

/* ---------- CAMERA ---------- */
const cam={x:0,y:0};

/* ---------- INPUT ---------- */
const keys={};
document.addEventListener("keydown",e=>{
  keys[e.key]=true;
  if(state==="title") state="play";
});
document.addEventListener("keyup",e=>keys[e.key]=false);

c.addEventListener("touchstart",e=>{
  const t=e.touches[0];
  if(t.clientX<c.width/2) keys["ArrowLeft"]=true;
  else keys["ArrowRight"]=true;
});
c.addEventListener("touchend",()=>{keys["ArrowLeft"]=keys["ArrowRight"]=false});

/* ---------- NPC ---------- */
const npcs=[
 {id:"old",x:900,y:1500,region:"village",name:"노인",
  talk:[
   {t:"해가 지면, 사람은 자기 이야기를 하게 되지.",flag:"old1"},
   {t:"길은 항상 있었어. 우리가 안 나갔을 뿐.",flag:"old2"}
  ]},
 {id:"woman",x:1800,y:1480,region:"village",name:"여인",
  talk:[
   {t:"숲 쪽으로 가봤어?",flag:"forest_hint"}
  ]},
 {id:"child",x:2600,y:1450,region:"village",name:"아이",
  talk:[
   {t:"저 멀리 반짝이는 게 있어.",flag:"gate_hint"}
  ]},
 {id:"hunter",x:3400,y:1550,region:"forest",name:"사냥꾼",
  talk:[
   {t:"여긴 길이 아니야. 선택이지.",flag:"hunter1"}
  ]},
 {id:"guard",x:4800,y:1500,region:"gate",name:"문지기",
  talk:[
   {t:"돌아갈 수도 있고, 넘어갈 수도 있다.",flag:"end_ready"}
  ]}
];

/* ---------- UTILS ---------- */
const clamp=(v,a,b)=>Math.max(a,Math.min(b,v));

/* ---------- UPDATE ---------- */
let time=0;
function update(){
  time+=0.04;
  if(state!=="play") return;

  if(!dialog){
    if(keys["ArrowLeft"]){hero.x-=hero.speed;hero.dir=-1;}
    if(keys["ArrowRight"]){hero.x+=hero.speed;hero.dir=1;}
    if(keys["ArrowUp"]){hero.y-=hero.speed;}
    if(keys["ArrowDown"]){hero.y+=hero.speed;}
  }

  hero.x=clamp(hero.x,60,world.w-60);
  hero.y=clamp(hero.y,200,world.h-60);

  cam.x+=(hero.x-cam.x-c.width/2)*0.08;
  cam.y+=(hero.y-cam.y-c.height/2)*0.08;

  checkNPC();
}

/* ---------- NPC CHECK ---------- */
function checkNPC(){
  dialog=null;
  npcs.forEach(n=>{
    if(n.region!==region) return;
    if(save.talked[n.id]) return;
    if(Math.abs(hero.x-n.x)<60 && Math.abs(hero.y-n.y)<60){
      dialog=n;
    }
  });
}

/* ---------- DRAW HERO ---------- */
function drawHero(){
  const x=hero.x-cam.x,y=hero.y-cam.y;
  const sway=Math.sin(time)*4;

  ctx.fillStyle="rgba(0,0,0,.25)";
  ctx.beginPath();ctx.ellipse(x,y+14,18,8,0,0,6.28);ctx.fill();

  ctx.fillStyle="#f0c94a";
  ctx.beginPath();
  ctx.moveTo(x-20,y-10);
  ctx.lineTo(x+10,y-10);
  ctx.lineTo(x+24,y+28);
  ctx.lineTo(x-30,y+28);
  ctx.closePath();ctx.fill();

  ctx.fillStyle="#3b3b3b";
  ctx.fillRect(x-8,y-6,16,20);

  ctx.fillStyle="#efe8db";
  ctx.beginPath();ctx.arc(x+hero.dir*2,y-22,9,0,6.28);ctx.fill();
}

/* ---------- DRAW ---------- */
function draw(){
  const r=regions[region];
  const g=ctx.createLinearGradient(0,0,0,c.height);
  g.addColorStop(0,r.sky1);g.addColorStop(1,r.sky2);
  ctx.fillStyle=g;ctx.fillRect(0,0,c.width,c.height);

  ctx.fillStyle=r.ground;
  ctx.fillRect(-cam.x,c.height-180-cam.y,world.w,260);

  npcs.forEach(n=>{
    if(n.region!==region) return;
    const x=n.x-cam.x,y=n.y-cam.y;
    ctx.fillStyle="#888";
    ctx.fillRect(x-6,y-20,12,20);
    ctx.fillStyle="#eee";
    ctx.beginPath();ctx.arc(x,y-28,7,0,6.28);ctx.fill();
  });

  drawHero();

  if(dialog){
    ctx.fillStyle="rgba(0,0,0,.6)";
    ctx.fillRect(80,c.height-140,c.width-160,80);
    ctx.fillStyle="#fff";
    ctx.font="16px serif";
    ctx.fillText(dialog.name+": "+dialog.talk[0].t,100,c.height-90);
  }

  if(state==="title"){
    ctx.fillStyle="rgba(0,0,0,.6)";
    ctx.fillRect(0,0,c.width,c.height);
    ctx.fillStyle="#fff";
    ctx.font="30px serif";
    ctx.fillText("A SMALL WORLD",c.width/2-120,c.height/2-10);
    ctx.font="14px serif";
    ctx.fillText("Tap / Press Any Key",c.width/2-80,c.height/2+20);
  }
}

/* ---------- INTERACT ---------- */
document.addEventListener("keydown",e=>{
  if(e.key===" " && dialog){
    save.talked[dialog.id]=true;
    saveGame();
    dialog=null;
    if(hero.x>3000) region="forest";
    if(hero.x>4400) region="gate";
    save.region=region;
  }
});

/* ---------- LOOP ---------- */
function loop(){update();draw();requestAnimationFrame(loop);}
loop();
</script>
