<div style="display:flex; justify-content:center; margin:40px 0;">
  <canvas id="game" width="900" height="500"
    style="background:#000; border-radius:12px;"></canvas>
</div>

<script>
/* =====================
   CORE
===================== */
const c = document.getElementById("game");
const ctx = c.getContext("2d");

/* =====================
   STATE
===================== */
let state = "title"; // title | play | end

/* =====================
   SAVE
===================== */
const save = JSON.parse(localStorage.getItem("edgeSave")) || {
  talked: {}
};
function saveGame(){
  localStorage.setItem("edgeSave", JSON.stringify(save));
}

/* =====================
   TIME & WEATHER
===================== */
let time = 0;
let weather = Math.random()<0.4 ? "rain" : Math.random()<0.5 ? "fog" : "clear";

/* =====================
   WORLD
===================== */
const world = { w:2000, h:1200 };

/* =====================
   PLAYER
===================== */
const player = { x:1000, y:600, speed:2 };

/* =====================
   CAMERA
===================== */
const cam = { x:0, y:0 };

/* =====================
   NPC
===================== */
const npcs = [
  { id:"old", x:800,y:700,name:"노인",
    text:["여긴 오래 머물 곳은 아니야.","그래도… 떠나긴 쉽지 않지."] },
  { id:"child", x:1200,y:500,name:"아이",
    text:["밖은 어떤 곳이야?","여긴 조금 작아."] },
  { id:"traveler", x:1400,y:800,name:"여행자",
    text:["이 길은 지도에 없어.","대신 기억에는 남지."] }
];

/* =====================
   INPUT
===================== */
const keys={};
document.addEventListener("keydown",e=>{
  keys[e.key]=true;
  if(state==="title") state="play";
});
document.addEventListener("keyup",e=>keys[e.key]=false);

/* =====================
   DIALOG
===================== */
let dialog=null;

/* =====================
   UPDATE
===================== */
function update(){
  if(state!=="play") return;

  time+=0.02;

  if(!dialog){
    if(keys["ArrowLeft"]) player.x-=player.speed;
    if(keys["ArrowRight"]) player.x+=player.speed;
    if(keys["ArrowUp"]) player.y-=player.speed;
    if(keys["ArrowDown"]) player.y+=player.speed;

    npcs.forEach(n=>{
      if(Math.hypot(player.x-n.x,player.y-n.y)<30 && keys[" "]){
        dialog={ npc:n, line:0 };
        save.talked[n.id]=true;
        saveGame();
      }
    });
  }

  // camera follow
  cam.x = Math.max(0, Math.min(player.x-450, world.w-900));
  cam.y = Math.max(0, Math.min(player.y-250, world.h-500));

  // ending condition
  if(Object.keys(save.talked).length===3 && time>200){
    state="end";
  }
}

/* =====================
   DRAW
===================== */
function drawSky(){
  let t = Math.min(time/200,1);
  let g = ctx.createLinearGradient(0,0,0,500);
  g.addColorStop(0, `rgb(${50-30*t},${70-40*t},${90-60*t})`);
  g.addColorStop(1, "#000");
  ctx.fillStyle=g;
  ctx.fillRect(0,0,900,500);
}

function draw(){
  ctx.clearRect(0,0,900,500);

  if(state==="title"){
    ctx.fillStyle="#000";ctx.fillRect(0,0,900,500);
    ctx.fillStyle="#fff";
    ctx.font="28px serif";
    ctx.fillText("AT THE EDGE OF EVENING",220,250);
    ctx.font="14px serif";
    ctx.fillText("press any key",390,300);
    return;
  }

  if(state==="end"){
    ctx.fillStyle="#000";ctx.fillRect(0,0,900,500);
    ctx.fillStyle="#fff";
    ctx.font="18px serif";
    ctx.fillText("You spoke to everyone.",340,240);
    ctx.fillText("The town did not stop you.",320,270);
    return;
  }

  drawSky();

  // ground
  ctx.fillStyle="#2c2b28";
  ctx.fillRect(-cam.x,800-cam.y,2000,400);

  // NPC
  npcs.forEach(n=>{
    ctx.fillStyle="#eee";
    ctx.beginPath();
    ctx.arc(n.x-cam.x,n.y-cam.y,6,0,6.28);
    ctx.fill();
  });

  // player
  ctx.fillStyle="#fff";
  ctx.beginPath();
  ctx.arc(player.x-cam.x,player.y-cam.y,6,0,6.28);
  ctx.fill();

  // weather
  if(weather==="rain"){
    ctx.strokeStyle="rgba(200,200,255,0.3)";
    for(let i=0;i<100;i++){
      ctx.beginPath();
      ctx.moveTo(Math.random()*900,Math.random()*500);
      ctx.lineTo(Math.random()*900+5,Math.random()*500+10);
      ctx.stroke();
    }
  }
  if(weather==="fog"){
    ctx.fillStyle="rgba(200,200,200,0.05)";
    ctx.fillRect(0,0,900,500);
  }

  // dialog
  if(dialog){
    ctx.fillStyle="rgba(0,0,0,0.7)";
    ctx.fillRect(100,360,700,100);
    ctx.fillStyle="#fff";
    ctx.font="14px serif";
    ctx.fillText(dialog.npc.name+": "+dialog.npc.text[dialog.line],120,400);
    ctx.fillText("1 계속   2 떠난다",120,430);

    if(keys["1"]){
      dialog.line++;
      if(dialog.line>=dialog.npc.text.length) dialog=null;
    }
    if(keys["2"]) dialog=null;
  }
}

/* =====================
   LOOP
===================== */
function loop(){
  update();
  draw();
  requestAnimationFrame(loop);
}
loop();
</script>
